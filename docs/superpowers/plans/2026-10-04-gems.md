# Gemmes supposées dans les châsses — plan d'implémentation

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Pawn compte chaque châsse avec la meilleure gemme bleue (rare) de l'extension du personnage (BC jusqu'au niveau 70, Wrath ensuite), avec des données de gemmes vérifiées contre le client.

**Architecture:** `tests/extract_gems.py` extrait les gemmes du client (stats, couleur) dans `tests/data/gems.frFR.txt`. Un test compare chaque gemme de `GemsWrath.lua` à ce fichier, et trois erreurs sont corrigées. `PawnWrathSetGemQualityForLevel` (fin de `GemsWrath.lua`) remplace `PawnGemQualityLevels` / `PawnMetaGemQualityLevels` par une seule entrée de niveau 0. `ClassicHawsJon.lua` l'appelle au chargement et sur `PLAYER_LEVEL_UP`. Au changement, toutes les échelles recalculent leurs meilleures gemmes, et `RatingWeightsVersion` passe à 2.

**Tech Stack:** Lua 5.1 (WoW 3.3.5a), LuaJIT pour les tests hors jeu, Python 3 + `mpyq` (via `uv`) pour l'extraction.

**Spec:** `docs/superpowers/specs/2026-10-04-gems-design.md`

```python
"""Extracts the gems of the frFR 3.3.5a client into tests/data/gems.frFR.txt: for each gem enchantment of
SpellItemEnchantment.dbc, the gem item, its color (GemProperties.dbc), its stat effects and its text.

Usage, from the addon root:
    uv run --with mpyq python tests/extract_gems.py "../../../Data"

Running it again on the same client data writes the same file, byte for byte.
"""
import hashlib
import struct
import sys

sys.dont_write_bytecode = True  # importing extract_ratings would otherwise leave tests/__pycache__ behind
from extract_ratings import read_file

ENCHANTS = "DBFilesClient\\SpellItemEnchantment.dbc"
GEMS = "DBFilesClient\\GemProperties.dbc"
OUTPUT = "tests/data/gems.frFR.txt"
# SpellItemEnchantment.dbc fields (32-bit, from 0): 0 ID, 2-4 effect type, 5-7 amount, 11-13 stat number (ITEM_MOD_*),
# 16 frFR text, 33 gem item (0 for enchantments that don't come from a gem).
ENCHANT_FIELDS = 38
# GemProperties.dbc fields: 1 enchantment, 4 color (1 meta, 2 red, 4 yellow, 8 blue; a gem of several colors adds them).
GEM_FIELDS = 5
STAT_EFFECT = 5  # ITEM_ENCHANTMENT_TYPE_STAT


def read_dbc(data_dir, name, fields):
    archive, data = read_file(data_dir, name)
    magic, records, count, record_size, _ = struct.unpack("<4s4i", data[:20])
    if magic != b"WDBC" or count != fields or record_size != fields * 4:
        sys.exit("format de %s inattendu : %r %d %d" % (name, magic, count, record_size))
    rows = [struct.unpack("<%di" % fields, data[20 + i * record_size:20 + (i + 1) * record_size]) for i in range(records)]
    return archive, data, rows, data[20 + records * record_size:]


def main():
    if len(sys.argv) != 2:
        sys.exit(__doc__)
    enchant_archive, enchant_data, enchants, strings = read_dbc(sys.argv[1], ENCHANTS, ENCHANT_FIELDS)
    gem_archive, gem_data, gems, _ = read_dbc(sys.argv[1], GEMS, GEM_FIELDS)
    color = {row[1]: row[4] for row in gems}

    def text(offset):
        return strings[offset:strings.index(b"\0", offset)].decode("utf-8")

    out = [
        "# Generated by tests/extract_gems.py: do not edit.",
        "# Source: %s from Data/%s (md5 %s)" % (ENCHANTS, enchant_archive, hashlib.md5(enchant_data).hexdigest()),
        "# Source: %s from Data/%s (md5 %s)" % (GEMS, gem_archive, hashlib.md5(gem_data).hexdigest()),
        "# SpellItemEnchantment fields: 2-4 effect type (5 = stat), 5-7 amount, 11-13 stat number, 16 text, 33 gem item.",
        "# GemProperties fields: 1 enchantment, 4 color (1 meta, 2 red, 4 yellow, 8 blue, added for several colors).",
        "# One line per gem enchantment: gem item<TAB>color<TAB>stat number:amount[,...]<TAB>text",
    ]
    for row in enchants:
        if row[33]:
            effects = ["%d:%d" % (row[11 + k], row[5 + k]) for k in range(3) if row[2 + k] == STAT_EFFECT]
            out.append("%d\t%d\t%s\t%s" % (row[33], color.get(row[0], 0), ",".join(effects), text(row[16])))
    with open(OUTPUT, "w", encoding="utf-8", newline="\n") as handle:
        handle.write("\n".join(out) + "\n")
    print("%s écrit (%d gemmes)" % (OUTPUT, len(out) - 6))


if __name__ == "__main__":
    main()
```

end

-- ITEM_MOD_* numbers of the stat effects of the GemsWrath.lua gems (tests/data/gems.frFR.txt), each attested by the text of a
-- one-stat gem enchantment: 3 "+6 Agilité" (2693), 4 "+6 Force" (2691), 5 "+6 Intelligence" (2694), 6 "+6 Esprit" (2699),
-- 7 "+9 Endurance" (2698), 12 "+6 au score de défense" (2696), 13 "+8 au score d'esquive" (2730), 14 "+8 au score de parade"
-- (2754), 31 "+6 au score de toucher" (2697), 32 "+6 au score de coup critique" (2695), 35 "+8 au score de résilience" (2759),
-- 36 "+8 au score de hâte" (3270), 37 "+12 au score d'expertise" (3379), 38 "+16 à la puissance d'attaque" (2729),
-- 43 "+3 points de mana toutes les 5 sec." (2701), 44 "+12 au score de pénétration d'armure" (3378),
-- 45 "+7 à la puissance des sorts" (2690).
local GemStatNumbers = { [3] = "Agility", [4] = "Strength", [5] = "Intellect", [6] = "Spirit", [7] = "Stamina",
	[12] = "DefenseRating", [13] = "DodgeRating", [14] = "ParryRating", [31] = "HitRating", [32] = "CritRating",
	[35] = "ResilienceRating", [36] = "HasteRating", [37] = "ExpertiseRating", [38] = "Ap", [43] = "Mp5",
	[44] = "ArmorPenetration", [45] = "SpellPower" }

Test("gemmes : chaque gemme de GemsWrath.lua a les stats et la couleur du client", function()
	assert(Gems70Rare and Meta70Rare, "tables de BC introuvables dans les listes d'origine")
	local Client = {}
	for Line in io.lines("tests/data/gems.frFR.txt") do
		local ID, Color, Effects = Line:match("^(%d+)\t(%d+)\t([%d:,]*)\t")
		if ID then
			ID = tonumber(ID)
			Client[ID] = Client[ID] or {}
			table.insert(Client[ID], { Color = tonumber(Color), Effects = Effects })
		end
	end
	local Failures, Checked = {}, 0
	local function Check(Levels, Meta)
		for _, Entry in ipairs(Levels) do
			for _, Gem in ipairs(Entry[2]) do
				local What = "gemme " .. Gem.ID
				local Lines = Client[Gem.ID]
				assert(Lines, What .. " absente de gems.frFR.txt")
				Equal(#Lines, 1, What .. " : enchantements")
				local Expected = {}
				for Number, Amount in Lines[1].Effects:gmatch("(%d+):(%d+)") do
					local Stat = GemStatNumbers[tonumber(Number)]
					assert(Stat, What .. " : numéro de stat " .. Number .. " inconnu")
					Expected[Stat] = (Expected[Stat] or 0) + tonumber(Amount)
				end
				local Got = {}
				for Stat, Amount in pairs(Gem.Stats) do
					-- Spell penetration is a spell effect (type 3) in SpellItemEnchantment.dbc, not a stat effect.
					if Stat ~= "SpellPenetration" then Got[Stat] = Amount end
				end
				local GotText, ExpectedText = Corpus.FormatStats(Got), Corpus.FormatStats(Expected)
				if GotText ~= ExpectedText then
					table.insert(Failures, What .. " : stats " .. GotText .. " au lieu de " .. ExpectedText)
				end
				local Color = (Gem.R and 2 or 0) + (Gem.Y and 4 or 0) + (Gem.B and 8 or 0)
				if not Meta and Color ~= Lines[1].Color then
					table.insert(Failures, What .. " : couleur " .. Color .. " au lieu de " .. Lines[1].Color)
				end
				Checked = Checked + 1
			end
		end
	end
	Check(WrathGemLevels, false)
	Check(WrathMetaGemLevels, true)
	assert(#Failures == 0, table.concat(Failures, " ; "))
	assert(Checked > 360, "gemmes vérifiées : " .. Checked)
end)

function PawnWrathSetGemQualityForLevel(Level)
	Level = max(1, min(80, floor(tonumber(Level) or 80)))
	local Quality = (Level <= 70) and 70 or 80
	if Quality == PawnWrathGemQuality then return false end
	PawnWrathGemQuality = Quality
	-- A single entry at item level 0: PawnGetGemQualityForItem gives it to every item.
	if Quality == 70 then
		PawnGemQualityLevels = { { 0, PawnGemData70Rare } }
		PawnMetaGemQualityLevels = { { 0, PawnMetaGemData70Rare } }
	else
		PawnGemQualityLevels = { { 0, PawnGemData80Rare } }
		PawnMetaGemQualityLevels = { { 0, PawnMetaGemData80Rare } }
	end
	return true
end
